```kupcad
plate = cube(x: 40, y: 30, z: 4, round_r: 2)
holes = [[5, 5], [35, 5], [5, 25], [35, 25]].map { |(x, y)| cylinder(d: 3.4, h: 10).translate(x, y, -1) }

plate.difference(holes)
```
